-- Drag the menu on mouse or touchscreen.
do
    local dragging, dragInput, dragStart, startPos
    Bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = Main.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    Bar.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input == dragInput or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

-- Aimlock: rotates toward the nearest living player.
local aimActive, aimConnection = false, nil
local function closestPlayer()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local chosen, dist = nil, math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local c = player.Character
            local r = c and c:FindFirstChild("HumanoidRootPart")
            local h = c and c:FindFirstChildOfClass("Humanoid")
            if r and h and h.Health > 0 then
                local d = (root.Position - r.Position).Magnitude
                if d < dist then chosen, dist = player, d end
            end
        end
    end
    return chosen
end
AimBtn.MouseButton1Click:Connect(function()
    aimActive = not aimActive
    setButtonState(AimBtn, "AIMLOCK", aimActive)
    if aimConnection then aimConnection:Disconnect(); aimConnection = nil end
    if aimActive then
        aimConnection = RunService.RenderStepped:Connect(function()
            local target = closestPlayer()
            local c = LocalPlayer.Character
            local r = c and c:FindFirstChild("HumanoidRootPart")
            local tc = target and target.Character
            local tr = tc and tc:FindFirstChild("HumanoidRootPart")
            if r and tr then local p,t = r.Position,tr.Position; r.CFrame = CFrame.new(p, Vector3.new(t.X,p.Y,t.Z)) end
        end)
    end
end)

-- ESP labels
local espActive, espObjects = false, {}
local function clearESP(player)
    if espObjects[player] then espObjects[player]:Destroy(); espObjects[player] = nil end
end
local function addESP(player, char)
    if player == LocalPlayer or not espActive or not char then return end
    clearESP(player)
    local head = char:FindFirstChild("Head") or char:WaitForChild("Head", 5)
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not head or not hum or not espActive then return end
    local gui = Instance.new("BillboardGui")
    gui.Name, gui.Adornee, gui.Size = "MiloPremiumESP", head, UDim2.new(0, 150, 0, 42)
    gui.StudsOffset, gui.AlwaysOnTop = Vector3.new(0, 2.4, 0), true
    gui.Parent = char
    local text = Instance.new("TextLabel")
    text.Size, text.BackgroundTransparency = UDim2.new(1,0,1,0), 1
    text.TextColor3, text.TextStrokeColor3, text.TextStrokeTransparency = NEON, Color3.new(0,0,0), 0.1
    text.Font, text.TextSize = Enum.Font.GothamBold, 12
    text.Parent = gui
    local function update() if gui.Parent and hum.Parent then text.Text = string.format("%s\nHP: %d/%d", player.Name, math.floor(hum.Health), math.floor(hum.MaxHealth)) end end
    update(); hum.HealthChanged:Connect(update)
    espObjects[player] = gui
end
local function hookPlayer(player)
    if player == LocalPlayer then return end
    player.CharacterAdded:Connect(function(char) if espActive then task.spawn(addESP, player, char) end end)
    if espActive and player.Character then task.spawn(addESP, player, player.Character) end
end
for _,p in ipairs(Players:GetPlayers()) do hookPlayer(p) end
Players.PlayerAdded:Connect(hookPlayer)
Players.PlayerRemoving:Connect(clearESP)
EspBtn.MouseButton1Click:Connect(function()
    espActive = not espActive; setButtonState(EspBtn, "ESP", espActive)
    if espActive then for _,p in ipairs(Players:GetPlayers()) do if p.Character then task.spawn(addESP,p,p.Character) end end
    else for p in pairs(espObjects) do clearESP(p) end end
end)

-- Fix lag: removes visual effects and disables shadows locally. This is destructive to client visuals.
FixBtn.MouseButton1Click:Connect(function()
    local classes = {ParticleEmitter=true, Trail=true, Smoke=true, Fire=true, Sparkles=true, Beam=true, Explosion=true, Highlight=true}
    for _,obj in ipairs(workspace:GetDescendants()) do
        if classes[obj.ClassName] then pcall(function() obj:Destroy() end)
        elseif obj:IsA("BasePart") then obj.CastShadow = false end
    end
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 100000
    FixBtn.Text = "FIX LAG: DONE"
end)

-- Toggle removal of tree-like parts as they appear.
local treesOn, treesConn = false, nil
local function isTree(obj)
    if not obj:IsA("BasePart") then return false end
    local n = string.lower(obj.Name)
    if n:find("leaf") or n:find("leaves") or n:find("bush") or n:find("plant") then return false end
    return n:find("tree") ~= nil or n:find("trunk") ~= nil or n:find("branch") ~= nil or (obj.Parent and string.lower(obj.Parent.Name):find("tree") ~= nil)
end
local function removeTreesNow()
    for _,obj in ipairs(workspace:GetDescendants()) do if not (LocalPlayer.Character and obj:IsDescendantOf(LocalPlayer.Character)) and isTree(obj) then pcall(function() obj:Destroy() end) end end
end
TreesBtn.MouseButton1Click:Connect(function()
    treesOn = not treesOn; setButtonState(TreesBtn, "XÓA CÂY", treesOn)
    if treesOn then removeTreesNow(); treesConn = workspace.DescendantAdded:Connect(function(obj) if treesOn and isTree(obj) then task.defer(function() if obj.Parent then obj:Destroy() end end) end end)
    elseif treesConn then treesConn:Disconnect(); treesConn = nil end
end)

-- Toggle removal of named trash/rock props.
local debrisOn, debrisConn = false, nil
local keywords = {"trash","bin","rock","stone","debris","rubble","dumpster","bench","chair","seat","ghế","spike","pillar","rockspike"}
local function isDebris(obj)
    if not (obj:IsA("BasePart") or obj:IsA("Model")) then return false end
    local n = string.lower(obj.Name)
    for _,word in ipairs(keywords) do if n:find(word,1,true) then return true end end
    return false
end
local function removeDebrisNow()
    for _,obj in ipairs(workspace:GetDescendants()) do if not (LocalPlayer.Character and obj:IsDescendantOf(LocalPlayer.Character)) and isDebris(obj) then pcall(function() obj:Destroy() end) end end
end
DebrisBtn.MouseButton1Click:Connect(function()
    debrisOn = not debrisOn; setButtonState(DebrisBtn, "XÓA RÁC & ĐÁ", debrisOn)
    if debrisOn then removeDebrisNow(); debrisConn = workspace.DescendantAdded:Connect(function(obj) if debrisOn and isDebris(obj) then task.defer(function() if obj.Parent then obj:Destroy() end end) end end)
    elseif debrisConn then debrisConn:Disconnect(); debrisConn = nil end
end)

-- Ping/FPS viewer toggle
local pingOn, pingGui, pingConn = false, nil, nil
local fps, frameCount, lastFps = 0, 0, os.clock()
RunService.RenderStepped:Connect(function()
    frameCount += 1
    local now = os.clock()
    if now-lastFps >= 1 then fps = math.floor(frameCount/(now-lastFps)); frameCount=0; lastFps=now end
end)
PingBtn.MouseButton1Click:Connect(function()
    pingOn = not pingOn; setButtonState(PingBtn,"PING VIEWER",pingOn)
    if pingOn then
        pingGui = Instance.new("TextLabel"); pingGui.Name="PingViewer"; pingGui.Size=UDim2.new(0,170,0,48); pingGui.Position=UDim2.new(1,-180,0,12)
        pingGui.BackgroundColor3=DARK; pingGui.BackgroundTransparency=0.12; pingGui.TextColor3=NEON; pingGui.TextStrokeTransparency=0.5
        pingGui.Font=Enum.Font.GothamBold; pingGui.TextSize=14; pingGui.Text="Ping: -- ms\nFPS: --"; pingGui.Parent=ScreenGui
        Instance.new("UICorner",pingGui).CornerRadius=UDim.new(0,8); local ps=Instance.new("UIStroke",pingGui); ps.Color=NEON; ps.Thickness=1.2
        pingConn = RunService.RenderStepped:Connect(function()
            if pingGui and pingGui.Parent then
                local ping = "N/A"
                pcall(function() local stats=game:GetService("Stats"); local item=stats.Network.ServerStatsItem["Data Ping"]; ping=math.floor(item:GetValue()).." ms" end)
                pingGui.Text = "Ping: "..ping.."\nFPS: "..tostring(fps)
            end
        end)
    else
        if pingConn then pingConn:Disconnect(); pingConn=nil end
        if pingGui then pingGui:Destroy(); pingGui=nil end
    end
end)
